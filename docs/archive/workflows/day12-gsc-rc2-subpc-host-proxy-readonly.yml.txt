name: Day12 GSC RC2 SubPC Authenticated Host READ ONLY Kit

on:
  workflow_dispatch:
  push:
    branches: [ main ]
    paths:
      - '.github/workflows/day12-gsc-rc2-subpc-host-proxy-readonly.yml'
      - 'tools/day12/Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.ps1'
      - 'tools/day12/Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.cmd'

permissions:
  contents: read

jobs:
  windows-fixtures:
    runs-on: windows-latest
    timeout-minutes: 8
    steps:
      - uses: actions/checkout@v7
      - name: PowerShell 5.1 syntax + read-only synthetic positive/negative fixtures
        shell: powershell
        run: |
          $p='tools/day12/Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.ps1'
          $tokens=$null;$errors=$null
          [void][Management.Automation.Language.Parser]::ParseFile((Resolve-Path $p),[ref]$tokens,[ref]$errors)
          if(@($errors).Count){throw 'Host proxy readonly check has parser errors'}
          $out=Join-Path $env:RUNNER_TEMP 'host-proxy-fixture'
          & powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -File $p -Synthetic -OutputDir $out
          if($LASTEXITCODE -ne 0){throw 'Host proxy synthetic scenario failed'}
          $reports=@(Get-ChildItem -LiteralPath $out -Filter '*.json' -File)
          if($reports.Count -ne 1){throw 'Host proxy fixture output mismatch'}
          $r=Get-Content -LiteralPath $reports[0].FullName -Raw -Encoding UTF8 | ConvertFrom-Json
          if($r.result -ne 'SYNTHETIC_PASS' -or !$r.synthetic -or !$r.read_only -or
             $r.mutation_performed -or $r.backend_ports_private -ne 'UNCHANGED_FAIL'){
            throw 'Host proxy fail-closed contract mismatch'
          }
      - name: Ensure only two GET endpoints and no credential file reads
        shell: pwsh
        run: |
          $p=Get-Content -LiteralPath 'tools/day12/Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.ps1' -Raw
          if($p -match '\.Method="POST"' -or $p -match 'ReadAllText\(.*client\.json' -or
             $p -match 'Authorization.*Bearer' -or $p -match 'Stop-Process|New-NetFirewallRule'){
            throw 'Forbidden operation in readonly Host proxy probe'
          }
          if(-not $p.Contains('"/api/health"') -or -not $p.Contains('"/api/status"')){
            throw 'Expected read-only endpoints missing'
          }
      - name: Assemble no-executable/no-install readonly handoff ZIP
        shell: pwsh
        run: |
          $out='host-proxy-kit'
          New-Item -ItemType Directory -Path $out -Force | Out-Null
          foreach($f in @('Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.cmd','Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.ps1')){
            Copy-Item -LiteralPath "tools/day12/$f" -Destination "$out/$f" -ErrorAction Stop
          }
          $readme=@'
          DAY12 GSC 4.3.9-rc.2 — SUB PC HOST CONNECTION (READ ONLY)
          ==========================================================
          The 06:31 local-runtime check already PASSED. Do not rerun it.
          On SUB PC only:
          1) Leave the GSC Client open as normal.
          2) Extract ZIP and run Day12_GSC_RC2_SubPC_Host_Proxy_READ_ONLY.cmd.
          3) Send the latest Desktop\Geumyi-Day12-SubPC-GSC\
             Day12-SubPC-GSC-RC2-HostProxy-*.json.
          4) If CHECK_REQUIRED, send the JSON; DO NOT edit pairing/tokens
             or reinstall/restore client without separate instructions.

          Only existing GSC Client local proxy 127.0.0.1:8790 is contacted
          via two HTTP GETs: /api/health and /api/status. The proxy uses the
          pre-existing pairing token for the authenticated Host request.
          No token/config/remote Host URL or server state is included in JSON.
          No commands, restart, update, service/firewall/world changes.
          PASS means read-only Host status is reachable using existing
          authentication, not that live Minecraft ports are private.
          backend_ports_private is still FAIL and Stable is blocked.
          '@
          [IO.File]::WriteAllText((Join-Path (Resolve-Path $out) 'README-FIRST.txt'),$readme,[Text.UTF8Encoding]::new($false))
          $members=@(Get-ChildItem -LiteralPath $out -File | Sort-Object Name)
          $checks=@($members|ForEach-Object {"$((Get-FileHash $_.FullName -Algorithm SHA256).Hash.ToLowerInvariant())  $($_.Name)"})
          [IO.File]::WriteAllLines((Join-Path (Resolve-Path $out) 'SHA256SUMS.txt'),$checks,[Text.Encoding]::ASCII)
          Compress-Archive -Path "$out/*" -DestinationPath 'Day12-GSC-RC2-SubPC-HOST-PROXY-READ_ONLY.zip' -CompressionLevel Optimal
          Add-Type -AssemblyName System.IO.Compression.FileSystem
          $z=[IO.Compression.ZipFile]::OpenRead((Resolve-Path 'Day12-GSC-RC2-SubPC-HOST-PROXY-READ_ONLY.zip'))
          try {
            if($z.Entries.Count -ne 4){throw 'Unexpected archive files'}
            foreach($e in $z.Entries){
              if(!$e.Name -or $e.Name -match '\.(exe|jar|dll)$'){throw 'Unsafe member'}
            }
          }finally {$z.Dispose()}
      - uses: actions/upload-artifact@v7
        with:
          name: day12-gsc-rc2-subpc-host-proxy-readonly-kit
          path: Day12-GSC-RC2-SubPC-HOST-PROXY-READ_ONLY.zip
          if-no-files-found: error
          retention-days: 14
