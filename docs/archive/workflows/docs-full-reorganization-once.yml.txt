name: One-time full documentation reorganization
# Retry after generated INDEX.md trailing-newline fix.

on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/docs-full-reorganization-once.yml'
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: geumyi-docs-once
  cancel-in-progress: false

jobs:
  migrate:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - name: Verify Python migration script compiles
        run: python3 -m py_compile tools/maintenance/organize_repo_docs.py
      - name: Print dry-run and protected source references
        run: python3 tools/maintenance/organize_repo_docs.py
      - name: Relocate historical documents and repair relative Markdown links
        run: python3 tools/maintenance/organize_repo_docs.py --apply
      - name: Fail on malformed Git changes
        run: |
          rm -rf tools/maintenance/__pycache__
          git add -A
          git diff --cached --check
          python3 - <<'PY'
          import subprocess
          lines=subprocess.check_output(['git','diff','--cached','--name-only'],text=True).splitlines()
          allowed=lambda p: p.endswith('.md') or p=='docs/README.md'
          invalid=[p for p in lines if not allowed(p)]
          if invalid:raise SystemExit('Unexpected non-Markdown change: '+repr(invalid))
          if not lines:raise SystemExit('No document changes; expected historic report relocation')
          print('Validated Markdown-only files:',len(lines))
          PY
      - name: Commit documentation-only changes
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git commit -m "docs: move remaining historical reports to day-wise and reference sections"
          git push origin HEAD:main
