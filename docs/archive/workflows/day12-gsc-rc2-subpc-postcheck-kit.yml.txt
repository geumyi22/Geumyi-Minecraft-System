name: Day12 GSC RC2 SubPC Runtime Postcheck READ ONLY Kit

on:
  workflow_dispatch:
  push:
    branches: [main]
    paths:
      - '.github/workflows/day12-gsc-rc2-subpc-postcheck-kit.yml'
      - 'tools/day12/Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.ps1'
      - 'tools/day12/Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.cmd'

permissions:
  contents: read

jobs:
  windows-readonly:
    runs-on: windows-latest
    timeout-minutes: 8
    steps:
      - uses: actions/checkout@v7
      - name: Parse PowerShell 5.1, run fail-closed synthetic fixtures, audit output
        shell: powershell
        run: |
          $path='tools/day12/Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.ps1'
          $tokens=$null;$errors=$null
          [void][Management.Automation.Language.Parser]::ParseFile((Resolve-Path $path),[ref]$tokens,[ref]$errors)
          if(@($errors).Count -ne 0){throw 'RC2 secondary PC postcheck parse errors'}
          $fixture=Join-Path $env:RUNNER_TEMP 'rc2-subpc-postcheck-fixture'
          & powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -File $path -Synthetic -OutputDir $fixture
          if($LASTEXITCODE -ne 0){throw 'Postcheck negative/positive synthetic fixture failed'}
          $entries=@(Get-ChildItem -LiteralPath $fixture -Filter '*.json' -File)
          if($entries.Count -ne 1){throw 'Expected one JSON output'}
          $report=Get-Content -LiteralPath $entries[0].FullName -Raw -Encoding UTF8|ConvertFrom-Json
          if($report.result -ne 'SYNTHETIC_PASS' -or !$report.synthetic -or !$report.read_only -or
             $report.system_mutated -or $report.backend_ports_private -ne 'UNCHANGED_FAIL'){
            throw 'Postcheck synthetic evidence violates safety contract'
          }
      - name: Package only readonly scripts, no client binaries
        shell: pwsh
        run: |
          $dest='postcheck-kit'
          New-Item -ItemType Directory -Path $dest -Force|Out-Null
          foreach($s in @('Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.ps1','Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.cmd')){
            Copy-Item -LiteralPath "tools/day12/$s" -Destination "$dest/$s"
          }
          $readme=@'
          DAY 12 — GSC 4.3.9-rc.2 SUB PC POST-UPDATE CHECK (READ ONLY)
          ============================================================
          On SUB PC only:
          1. If GSC is closed, start GSC Client normally from Start menu.
          2. Unzip and double-click Day12_GSC_RC2_SubPC_Postcheck_READ_ONLY.cmd.
          3. Send latest Desktop\Geumyi-Day12-SubPC-GSC\
             Day12-SubPC-GSC-RC2-Postcheck-*.json.
          4. CHECK_REQUIRED: send JSON without doing any repair/reinstall.
          5. Do NOT run restore 4.3.8 yet; wait until actual UI/host
             connectivity is checked and the next action is specified.

          Reads Client executable SHA, process identity, localhost /app.ico,
          latest sanitized client-update helper report and existing official
          4.3.8 backup SHA. May try native TCP owner lookup as an OPTIONAL
          cross-check, not an authoritative backend or IPv6 bind attestation.
          Does NOT start/stop clients, run updates, read tokens/client.json,
          modify any config/registry/firewall/host/world/backup or send secrets.
          PASS is local client health only. Remote Host API/GSCM connection,
          OS private backend binding and Day 12 Stable gates remain separate.
          '@
          [IO.File]::WriteAllText((Join-Path (Resolve-Path $dest) 'README-FIRST.txt'),$readme,[Text.UTF8Encoding]::new($false))
          $fileList=@(Get-ChildItem $dest -File|Sort-Object Name)
          $hashes=@($fileList|ForEach-Object{"$((Get-FileHash $_.FullName -Algorithm SHA256).Hash.ToLowerInvariant())  $($_.Name)"})
          [IO.File]::WriteAllLines((Join-Path (Resolve-Path $dest) 'SHA256SUMS.txt'),$hashes,[Text.Encoding]::ASCII)
          Compress-Archive -Path "$dest/*" -DestinationPath 'Day12-GSC-RC2-SubPC-POSTCHECK-READ_ONLY.zip' -CompressionLevel Optimal
          Add-Type -AssemblyName System.IO.Compression.FileSystem
          $z=[IO.Compression.ZipFile]::OpenRead((Resolve-Path 'Day12-GSC-RC2-SubPC-POSTCHECK-READ_ONLY.zip'))
          try {
            if($z.Entries.Count -ne 4){throw 'Unexpected ZIP entries'}
            foreach($entry in $z.Entries){
              if(!$entry.Name -or $entry.Name -match '\.(exe|jar|dll)$'){throw 'Executable or invalid member'}
            }
          }finally{$z.Dispose()}
          Write-Host 'READ_ONLY_POSTCHECK_KIT_ASSEMBLED'
      - uses: actions/upload-artifact@v7
        with:
          name: day12-gsc-rc2-subpc-postcheck-readonly-kit
          path: Day12-GSC-RC2-SubPC-POSTCHECK-READ_ONLY.zip
          if-no-files-found: error
          retention-days: 14
