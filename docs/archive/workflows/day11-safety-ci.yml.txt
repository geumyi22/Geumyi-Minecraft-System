name: Day 11 Safety Preflight CI

on:
  push:
    paths:
      - 'tools/day11/**'
      - 'DAY11-PLAN.md'
      - '.github/workflows/day11-safety-ci.yml'
  pull_request:
    branches: [ main ]
    paths:
      - 'tools/day11/**'
      - 'DAY11-PLAN.md'
      - 'GSC/ServerCenter/**'
      - 'GSCM/**'
      - '.github/workflows/day11-safety-ci.yml'
  workflow_dispatch:

permissions:
  contents: read

jobs:
  readonly-preflight:
    name: Read-only preflight safety guard
    runs-on: windows-latest
    timeout-minutes: 10
    steps:
      - name: Checkout
        uses: actions/checkout@v7

      - name: Parse PowerShell
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $path = 'tools/day11/Day11_Phase0_ReadOnly_Preflight.ps1'
          $tokens = $null
          $errors = $null
          [void][System.Management.Automation.Language.Parser]::ParseFile(
            (Resolve-Path $path),
            [ref]$tokens,
            [ref]$errors
          )
          if($errors.Count -gt 0){
            $errors | Format-List | Out-String | Write-Host
            throw 'Day 11 preflight PowerShell parse failed'
          }

      - name: Parse Phase 7 protection verifier
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $path = 'tools/day11/Day11_Phase7_Protection_READ_ONLY.ps1'
          $tokens = $null
          $errors = $null
          [void][System.Management.Automation.Language.Parser]::ParseFile(
            (Resolve-Path $path),
            [ref]$tokens,
            [ref]$errors
          )
          if($errors.Count -gt 0){
            $errors | Format-List | Out-String | Write-Host
            throw 'Day 11 Phase 7 PowerShell parse failed'
          }

      - name: Reject destructive Phase 7 verifier commands
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $raw = Get-Content -LiteralPath 'tools/day11/Day11_Phase7_Protection_READ_ONLY.ps1' -Raw
          $forbidden = @(
            '(?i)\bStop-Process\b',
            '(?i)\btaskkill(?:\.exe)?\b',
            '(?i)\bRemove-Item\b',
            '(?i)\bMove-Item\b',
            '(?i)\bCopy-Item\b',
            '(?i)\bRestart-Computer\b',
            '(?i)\bStop-Computer\b',
            '(?i)\bStop-Service\b',
            '(?i)\bStart-Service\b',
            '(?i)DELETE_PERMANENT_BACKUP',
            '(?i)MOVE_TO_TRASH',
            '(?i)/api/v[14]/servers/.+/(actions|schedule)',
            '(?i)/api/v4/restore["'']',
            '(?i)Post-Json\s+["'']/api/v4/backup["'']'
          )
          foreach($pattern in $forbidden){
            if($raw -match $pattern){
              throw "Phase 7 READ ONLY verifier contains forbidden mutation pattern: $pattern"
            }
          }
          if($raw -notmatch '__day11_phase7_confirmation_probe_DOES_NOT_EXIST__'){
            throw 'Permanent-delete confirmation probe must use the impossible disposable filename'
          }
          if($raw -notmatch 'mutation_performed\s*=\s*\$false'){
            throw 'Phase 7 report must explicitly declare mutation_performed=false'
          }

      - name: Parse disposable Phase 7 backup E2E
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $path = 'tools/day11/Day11_Phase7_Disposable_Backup_E2E.ps1'
          $tokens = $null
          $errors = $null
          [void][System.Management.Automation.Language.Parser]::ParseFile(
            (Resolve-Path $path),
            [ref]$tokens,
            [ref]$errors
          )
          if($errors.Count -gt 0){
            $errors | Format-List | Out-String | Write-Host
            throw 'Day 11 disposable Phase 7 PowerShell parse failed'
          }

      - name: Guard disposable Phase 7 backup E2E boundary
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $raw = Get-Content -LiteralPath 'tools/day11/Day11_Phase7_Disposable_Backup_E2E.ps1' -Raw
          $forbidden = @(
            '(?i)\bStop-Process\b',
            '(?i)\btaskkill(?:\.exe)?\b',
            '(?i)\bRemove-Item\b',
            '(?i)\bMove-Item\b',
            '(?i)\bCopy-Item\b',
            '(?i)\bRestart-Computer\b',
            '(?i)\bStop-Computer\b',
            '(?i)\bStop-Service\b',
            '(?i)\bStart-Service\b',
            '(?i)DELETE_PERMANENT_BACKUP',
            '(?i)/api/v[14]/servers/.+/(actions|schedule)',
            '(?i)Post-Json\s+["'']/api/v4/restore["'']'
          )
          foreach($pattern in $forbidden){
            if($raw -match $pattern){
              throw "Disposable Phase 7 E2E escaped backup-only safety boundary: $pattern"
            }
          }
          if($raw -notmatch 'scope\s*=\s*["'']config["'']'){
            throw 'Disposable E2E must create config-scope backup only'
          }
          if($raw -notmatch 'minecraft_data_restore_performed\s*=\s*\$false'){
            throw 'Disposable E2E must report Minecraft data restore=false'
          }
          if($raw -notmatch 'server_lifecycle_action_performed\s*=\s*\$false'){
            throw 'Disposable E2E must report server lifecycle action=false'
          }

      - name: Parse Day 11 final read-only E2E
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $path = 'tools/day11/Day11_Final_E2E_READ_ONLY.ps1'
          $tokens = $null
          $errors = $null
          [void][System.Management.Automation.Language.Parser]::ParseFile(
            (Resolve-Path $path),
            [ref]$tokens,
            [ref]$errors
          )
          if($errors.Count -gt 0){
            $errors | Format-List | Out-String | Write-Host
            throw 'Day 11 final E2E PowerShell parse failed'
          }

      - name: Enforce Day 11 final E2E read-only boundary
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $raw = Get-Content -LiteralPath 'tools/day11/Day11_Final_E2E_READ_ONLY.ps1' -Raw
          $forbidden = @(
            '(?i)\bStop-Process\b',
            '(?i)\btaskkill(?:\.exe)?\b',
            '(?i)\bRemove-Item\b',
            '(?i)\bMove-Item\b',
            '(?i)\bCopy-Item\b',
            '(?i)\bRestart-Computer\b',
            '(?i)\bStop-Computer\b',
            '(?i)\bStop-Service\b',
            '(?i)\bStart-Service\b',
            '(?i)DELETE_PERMANENT_BACKUP',
            '(?i)MOVE_TO_TRASH',
            '(?i)/api/v1/servers/.+/(actions|schedule|command|player-action)',
            '(?i)/api/server/action',
            '(?i)/api/v4/update/(decision|policy|external/stage|external/apply-proxy|self/stage|self/apply)',
            '(?i)/api/v4/backup/action',
            '(?i)/api/v4/backup/retention/apply',
            '(?i)Post-Json\s+["'']/api/v4/restore["'']'
          )
          foreach($pattern in $forbidden){
            if($raw -match $pattern){
              throw "Day 11 final E2E contains forbidden mutation path: $pattern"
            }
          }

          $allowedPost = @(
            '/api/v4/update/dry-run',
            '/api/v4/backup/retention/dry-run',
            '/api/v4/restore/preflight'
          )
          $matches = [regex]::Matches($raw, 'Post-Json\s+["'']([^"'']+)["'']')
          foreach($m in $matches){
            $endpoint = [string]$m.Groups[1].Value
            if($allowedPost -notcontains $endpoint){
              throw "Day 11 final E2E POST endpoint is not explicitly read-only: $endpoint"
            }
          }
          foreach($flag in @(
            'mutation_performed = $false',
            'update_stage_or_apply_performed = $false',
            'policy_change_performed = $false',
            'server_lifecycle_action_performed = $false',
            'minecraft_data_restore_performed = $false',
            'backup_move_or_delete_performed = $false'
          )){
            if($raw -notlike ('*' + $flag + '*')){
              throw "Missing final E2E safety declaration: $flag"
            }
          }

      - name: Reject destructive preflight commands
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $raw = Get-Content -LiteralPath 'tools/day11/Day11_Phase0_ReadOnly_Preflight.ps1' -Raw
          $forbidden = @(
            '(?i)\bStop-Process\b',
            '(?i)\btaskkill(?:\.exe)?\b',
            '(?i)\bRemove-Item\b',
            '(?i)\bMove-Item\b',
            '(?i)\bCopy-Item\b',
            '(?i)\bNew-NetFirewallRule\b',
            '(?i)\bRemove-NetFirewallRule\b',
            '(?i)\bSet-NetFirewallRule\b',
            '(?i)\bRestart-Computer\b',
            '(?i)\bStop-Computer\b',
            '(?i)\bStop-Service\b',
            '(?i)\bStart-Service\b',
            '(?i)\bSet-Service\b',
            '(?i)Invoke-RestMethod[^\r\n]*-Method\s+(Post|Put|Delete|Patch)'
          )
          foreach($pattern in $forbidden){
            if($raw -match $pattern){
              throw "READ ONLY preflight contains forbidden mutation pattern: $pattern"
            }
          }

      - name: Execute synthetic read-only preflight
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $out = Join-Path $env:RUNNER_TEMP 'day11-preflight'
          & powershell.exe -NoProfile -ExecutionPolicy Bypass -File 'tools/day11/Day11_Phase0_ReadOnly_Preflight.ps1' -NoNetworkProbe -OutputDir $out
          if($LASTEXITCODE -ne 0){ throw "preflight exit code $LASTEXITCODE" }
          $json = Get-ChildItem -LiteralPath $out -Filter 'Geumyi-Day11-Phase0-*.json' | Sort-Object LastWriteTime -Descending | Select-Object -First 1
          if(-not $json){ throw 'preflight JSON missing' }
          $r = Get-Content -LiteralPath $json.FullName -Raw | ConvertFrom-Json
          if($r.phase -ne 'Day11-Phase0-READ-ONLY'){ throw 'wrong phase marker' }
          foreach($name in @('mutates_server_config','stops_minecraft','stops_velocity','changes_firewall','reads_secrets')){
            if([bool]$r.safety.$name){ throw "unsafe preflight flag: $name" }
          }
          Write-Host 'DAY 11 READ-ONLY PREFLIGHT SAFETY CI PASS'
