name: Day12 GSC Canary Draft Read-Only Audit

on:
  workflow_dispatch:
  push:
    branches: [ main ]
    paths:
      - '.github/workflows/day12-gsc-canary-draft-audit.yml'

# Draft releases can be hidden from read-only GitHub Actions tokens.
# Grant release visibility to the audit job; script performs GET only.
permissions:
  contents: write

jobs:
  audit:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    env:
      GH_TOKEN: ${{ github.token }}
      TAG: system-2026.10.10-day12-gsc439-rc1-canary
    steps:
      - name: Read release-list API including DRAFTS (never modify)
        shell: bash
        run: |
          set -euo pipefail
          gh api --paginate "repos/$GITHUB_REPOSITORY/releases?per_page=100" > "$RUNNER_TEMP/releases.json"
          python3 - <<'PY'
          import json,os,pathlib
          p=pathlib.Path(os.environ["RUNNER_TEMP"])/"releases.json"
          raw=p.read_text()
          try:
              arr=json.loads(raw)
          except json.JSONDecodeError:
              print("Release list unexpectedly paginated; manual recheck required")
              raise
          tag=os.environ["TAG"]
          xs=[r for r in arr if r.get("tag_name")==tag]
          # GitHub may assign an 'untagged-...' placeholder internally while
          # a draft has never been published. Match exact title and source.
          if not xs:
              xs=[r for r in arr if r.get("draft") is True
                  and r.get("name")=="GSC 4.3.9-rc.1 Canary — DRAFT / NOT LIVE"
                  and r.get("target_commitish")=="2e5bec5f4fa0275f6bef6266f7772de67e7a14b3"]
          print("candidate-draft-count:",len(xs))
          if not xs:
              print("last-five-drafts:",[(x.get("tag_name"), x.get("name"), len(x.get("assets",[])))
                     for x in arr if x.get("draft")][:5])
          assert len(xs)==1, "Expected exactly one candidate draft by pin or exact title/source"
          r=xs[0]
          names=sorted(a["name"] for a in r["assets"])
          print("draft:",r["draft"],"prerelease:",r["prerelease"],
                "asset-count:",len(names))
          print("asset-names:",", ".join(names))
          assert r["draft"] is True
          assert r["prerelease"] is True
          assert r["published_at"] is None
          for name in [
              "GeumyiServerCenter-v4.3.9-rc.1-Setup.exe",
              "deployment-canary.json",
              "deployment-canary.json.sig",
              "deployment-public.pem",
              "SHA256SUMS.txt",
              "README-NOT-FOR-INSTALL.txt",
          ]:
              assert name in names, "Missing draft attachment: "+name
          assert "deployment-stable.json" not in names
          assert r.get("target_commitish")=="2e5bec5f4fa0275f6bef6266f7772de67e7a14b3"
          pathlib.Path(os.environ["RUNNER_TEMP"],"draft_id.txt").write_text(str(r["id"]))
          print("CANARY_DRAFT_ONLY_AUDIT_PASS")
          PY

      - name: Independently re-download DRAFT assets and verify manifest signature
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p "$RUNNER_TEMP/canary-draft-readback"
          python3 - <<'PY' > "$RUNNER_TEMP/assets.tsv"
          import json,os,pathlib
          releases=json.loads((pathlib.Path(os.environ["RUNNER_TEMP"])/"releases.json").read_text())
          rid=int((pathlib.Path(os.environ["RUNNER_TEMP"])/"draft_id.txt").read_text())
          r=next(r for r in releases if r.get("id")==rid)
          expected={
              "GeumyiServerCenter-v4.3.9-rc.1-Setup.exe",
              "deployment-canary.json",
              "deployment-canary.json.sig",
              "deployment-public.pem",
              "SHA256SUMS.txt",
              "README-NOT-FOR-INSTALL.txt",
          }
          assert {a["name"] for a in r["assets"]} == expected
          for a in sorted(r["assets"],key=lambda a:a["name"]):
              assert int(a["size"]) > 0
              print(f'{a["name"]}\t{a["id"]}')
          PY
          while IFS=$'\t' read -r name id; do
            case "$name" in
              GeumyiServerCenter-v4.3.9-rc.1-Setup.exe|deployment-canary.json|deployment-canary.json.sig|deployment-public.pem|SHA256SUMS.txt|README-NOT-FOR-INSTALL.txt) ;;
              *) echo "::error::Unexpected draft asset"; exit 1 ;;
            esac
            gh api -H 'Accept: application/octet-stream' \
              "repos/$GITHUB_REPOSITORY/releases/assets/$id" \
              > "$RUNNER_TEMP/canary-draft-readback/$name"
          done < "$RUNNER_TEMP/assets.tsv"
          cd "$RUNNER_TEMP/canary-draft-readback"
          sha256sum --check SHA256SUMS.txt
          echo '5d6a69ade04f3afb1c1e5367b5380559cadcf901eda35eb383d6cb83ed2ec539  GeumyiServerCenter-v4.3.9-rc.1-Setup.exe' | sha256sum --check
          openssl pkeyutl -verify -rawin -pubin \
            -inkey deployment-public.pem \
            -in deployment-canary.json \
            -sigfile deployment-canary.json.sig
          python3 - <<'PY'
          import json
          m=json.load(open("deployment-canary.json",encoding="utf-8"))
          assert m["schema"]==1 and m["channel"]=="canary"
          assert m["repository"]=="geumyi22/Geumyi-Minecraft-System"
          assert m["release"]=="system-2026.10.10-day12-gsc439-rc1-canary"
          assert set(m["components"])=={"gsc"}
          c=m["components"]["gsc"]
          assert c["version"]=="4.3.9-rc.1"
          assert c["kind"]=="gsc" and c["targets"]==["host"]
          assert c["sha256"]=="5d6a69ade04f3afb1c1e5367b5380559cadcf901eda35eb383d6cb83ed2ec539"
          print("SIGNED_GSC_CANARY_DRAFT_ASSET_READBACK_PASS")
          PY
