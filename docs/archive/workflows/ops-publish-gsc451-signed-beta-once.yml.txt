name: Publish signed GSC 4.5.1 Beta update (GSC only; one-time)
on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/ops-publish-gsc451-signed-beta-once.yml'
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: geumyi-gsc451-signed-beta-once
  cancel-in-progress: false

jobs:
  signed-beta:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    env:
      GH_TOKEN: ${{ github.token }}
      DEPLOYMENT_PRIVATE_KEY_B64: ${{ secrets.DEPLOYMENT_ED25519_PRIVATE_KEY_B64 }}
      SOURCE_TAG: system-2026.10.11-stable-gsc451-gscm151
      SOURCE_COMMIT: 06bfbe53c53d966331834ecd8301f4764df60679
      TARGET_TAG: system-2026.10.11-gsc451-signed-beta.1
      INSTALLER: GeumyiServerCenter-v4.5.1-Setup.exe
      INSTALLER_SHA256: 9b5e79f10cfd7eb0080cbeaaad0339c531804e393b0a897377c411f185c588da
      PRIOR_SIGNED_BETA: system-2026.10.07-day11-gsc438-beta
    steps:
      - name: Checkout manifest generator from protected main
        uses: actions/checkout@v7
      - name: Refuse missing signing secret or duplicate tag
        shell: bash
        run: |
          set -euo pipefail
          [[ -n "$DEPLOYMENT_PRIVATE_KEY_B64" ]] || { echo "::error::Deployment signing secret is not configured"; exit 1; }
          if gh release view "$TARGET_TAG" >/dev/null 2>&1; then echo "::error::Already published"; exit 1; fi
          existing="$(git ls-remote --tags --refs origin "refs/tags/$TARGET_TAG")"
          [[ -z "$existing" ]] || { echo "::error::Ref already exists"; exit 1; }
      - name: Verify pinned source release metadata and asset digest
        shell: bash
        run: |
          set -euo pipefail
          gh api "repos/$GITHUB_REPOSITORY/releases/tags/$SOURCE_TAG" > source-release.json
          python3 - <<'PY'
          import os,json
          r=json.load(open("source-release.json"))
          assert not r["draft"] and not r["prerelease"]
          assert r["tag_name"]==os.environ["SOURCE_TAG"]
          assert r["target_commitish"]==os.environ["SOURCE_COMMIT"]
          matching=[a for a in r["assets"] if a["name"]==os.environ["INSTALLER"]]
          assert len(matching)==1
          a=matching[0]
          assert a["state"]=="uploaded" and a["size"]>20000000
          assert a.get("digest")=="sha256:"+os.environ["INSTALLER_SHA256"]
          print("Pinned source package digest verified")
          PY
      - name: Fetch immutable GSC Setup and known-good Beta public key
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p release-assets original-beta
          gh release download "$SOURCE_TAG" --repo "$GITHUB_REPOSITORY" --pattern "$INSTALLER" --dir release-assets
          (cd release-assets && echo "$INSTALLER_SHA256  $INSTALLER" | sha256sum -c -)
          gh release download "$PRIOR_SIGNED_BETA" --repo "$GITHUB_REPOSITORY" --pattern 'deployment-public.pem' --dir original-beta
      - name: Verify this signing secret matches already trusted Beta key
        shell: bash
        run: |
          set -euo pipefail
          printf '%s' "$DEPLOYMENT_PRIVATE_KEY_B64" | base64 --decode > "$RUNNER_TEMP/deployment-private.pem"
          chmod 600 "$RUNNER_TEMP/deployment-private.pem"
          openssl pkey -in "$RUNNER_TEMP/deployment-private.pem" -check -noout
          openssl pkey -in "$RUNNER_TEMP/deployment-private.pem" -pubout -out "$RUNNER_TEMP/from-secret-public.pem"
          openssl pkey -pubin -in original-beta/deployment-public.pem -pubout -outform DER > "$RUNNER_TEMP/original-pub.der"
          openssl pkey -pubin -in "$RUNNER_TEMP/from-secret-public.pem" -pubout -outform DER > "$RUNNER_TEMP/new-pub.der"
          cmp "$RUNNER_TEMP/original-pub.der" "$RUNNER_TEMP/new-pub.der"
          echo "Deployment signer matches previously released Beta trust key"
      - name: Generate manifest containing only GSC Setup (no plugin auto-apply)
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p "$RUNNER_TEMP/beta-catalog"
          python3 - <<'PY'
          import json, pathlib, os
          c=json.loads(pathlib.Path('deploy/components.json').read_text(encoding='utf-8'))
          g=c['components']['gsc']
          assert g['version']=='4.5.1' and g['kind']=='gsc' and g['artifact_glob']==os.environ['INSTALLER']
          assert 'host' in g['targets']
          c['components']={'gsc':g}
          pathlib.Path(os.environ['RUNNER_TEMP'],'beta-catalog','components.json').write_text(json.dumps(c,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
          PY
          python3 tools/release/generate_manifest.py \
            --catalog "$RUNNER_TEMP/beta-catalog/components.json" \
            --assets release-assets \
            --channel beta \
            --tag "$TARGET_TAG" \
            --base-url "https://github.com/$GITHUB_REPOSITORY/releases/download/$TARGET_TAG" \
            --output release-assets/deployment-beta.json
          python3 - <<'PY'
          import json,os
          m=json.load(open('release-assets/deployment-beta.json'))
          assert m['channel']=='beta' and m['release']==os.environ['TARGET_TAG']
          assert set(m['components'])=={'gsc'}
          assert m['components']['gsc']['sha256']==os.environ['INSTALLER_SHA256']
          assert m['components']['gsc']['version']=='4.5.1'
          PY
      - name: Ed25519 sign and verify using old pinned public key
        shell: bash
        run: |
          set -euo pipefail
          openssl pkeyutl -sign -rawin -inkey "$RUNNER_TEMP/deployment-private.pem" \
            -in release-assets/deployment-beta.json \
            -out release-assets/deployment-beta.json.sig
          openssl pkeyutl -verify -rawin -pubin \
            -inkey original-beta/deployment-public.pem \
            -in release-assets/deployment-beta.json \
            -sigfile release-assets/deployment-beta.json.sig
          (cd release-assets && sha256sum deployment-beta.json > deployment-beta.json.sha256)
          cp original-beta/deployment-public.pem release-assets/deployment-public.pem
          (cd release-assets && sha256sum "$INSTALLER" deployment-beta.json deployment-beta.json.sig deployment-public.pem > SHA256SUMS.txt)
          cat > release-assets/RELEASE-NOTES.txt <<'TXT'
          GSC 4.5.1 signed Beta update availability only.
          Verifies with the established Day-11 Beta Ed25519 public key.
          Hosts on channel=beta will discover this as a newer signed GSC Setup.
          An operator must explicitly stage/apply the Host and Client update.
          No server plugin, world, backup, firewall, or GSCM APK is auto-applied by this manifest.
          This is NOT approval for an automatic Stable rollout.
          TXT
      - name: Publish immutable signed Beta prerelease
        shell: bash
        run: |
          set -euo pipefail
          gh release create "$TARGET_TAG" release-assets/* \
            --repo "$GITHUB_REPOSITORY" \
            --target "$SOURCE_COMMIT" \
            --title "GSC 4.5.1 — Signed Beta GSC Auto-Update Channel" \
            --prerelease \
            --notes "GSC 4.5.1 only: Ed25519-signed deployment-beta.json with verified installer SHA-256 and the same public key as the Day-11 signed Beta. On Beta, GSC can discover/stage the new version; Host/Client apply requires explicit user action. No plugins or Minecraft server processes are updated. Formal Stable auto-deployment remains blocked."
      - name: Verify public release and exact seven assets
        shell: bash
        run: |
          set -euo pipefail
          gh api "repos/$GITHUB_REPOSITORY/releases/tags/$TARGET_TAG" > published.json
          python3 - <<'PY'
          import json,os,re
          p=json.load(open('published.json'))
          expected={os.environ['INSTALLER'],'deployment-beta.json','deployment-beta.json.sig',
                    'deployment-beta.json.sha256','deployment-public.pem',
                    'SHA256SUMS.txt','RELEASE-NOTES.txt'}
          assert p['draft'] is False and p['prerelease'] is True
          assert p['tag_name']==os.environ['TARGET_TAG']
          assert {a['name'] for a in p['assets']}==expected
          assert all(a['state']=='uploaded' and a.get('digest') and a['size']>0 for a in p['assets'])
          print('Published verified signed Beta:',p['html_url'])
          PY
